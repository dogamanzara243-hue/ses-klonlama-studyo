import gradio as gr, edge_tts

async def sesleri_getir():
    tum_sesler = await edge_tts.list_voices()
    tr = [s for s in tum_sesler if s["Locale"].startswith("tr-")]
    diger = [s for s in tum_sesler if not s["Locale"].startswith("tr-")]
    return [{"id": s["ShortName"], "ad": f"{s['ShortName']} ({'Kadın' if s['Gender']=='Female' else 'Erkek'} - {s['Locale']})"} for s in (tr + diger)]

async def ses_uret(metin, ses, hiz):
    dosya = "seslendirme.mp3"
    await edge_tts.Communicate(metin.strip(), ses, rate=hiz).save(dosya)
    return dosya

with gr.Blocks() as demo:
    m, s, h = gr.Textbox(), gr.Textbox(value="tr-TR-AhmetNeural"), gr.Textbox(value="+5%")
    gr.Button("Üret").click(fn=ses_uret, inputs=[m, s, h], outputs=gr.Audio(type="filepath"), api_name="ses_uret")
    gr.Button("Liste").click(fn=sesleri_getir, outputs=gr.JSON(), api_name="sesleri_getir")

demo.launch(share=True)
